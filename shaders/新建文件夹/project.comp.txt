#version 450

layout(local_size_x = 128, local_size_y = 1, local_size_z = 1) in;

const uint NX = 128u;
const uint NY = 128u;
const float DX = 1.0f / float(NX);
const float INV_2DX = 0.5f / DX;

// 边界偏移常量
const uint TOP_START    = 0u;
const uint BOTTOM_START = NX;
const uint LEFT_START   = 2u * NX;
const uint RIGHT_START  = 2u * NX + NY;

layout(std430, binding = 0) buffer PressureBuffer { float data[]; } p;
layout(std430, binding = 2) buffer UBuffer { float data[]; } u;
layout(std430, binding = 3) buffer VBuffer { float data[]; } v;
layout(std430, binding = 5) buffer BCType  { int data[]; } bc_type;
layout(std430, binding = 6) buffer BCVelU  { float data[]; } bc_vel_u;   // 本求解器中未使用
layout(std430, binding = 7) buffer BCVelV  { float data[]; } bc_vel_v;   // 本求解器中未使用

float getPressure(int i, int j, uint self_idx) {
    if (i >= 0 && i < int(NX) && j >= 0 && j < int(NY)) {
        uint idx = uint(i) + uint(j) * NX;
        return p.data[idx];
    }
    // 越界，查询边界类型
    int dir = -1;
    uint local = 0;
    if (i < 0) { dir = 0; local = uint(j); }
    else if (i >= int(NX)) { dir = 2; local = uint(j); }
    else if (j < 0) { dir = 3; local = uint(i); }
    else { dir = 1; local = uint(i); }

    uint offset = (dir == 0) ? LEFT_START : (dir == 2) ? RIGHT_START : (dir == 1) ? TOP_START : BOTTOM_START;
    int t = bc_type.data[offset + local];
    if (t == 1) {               // NoSlipWall → Neumann: 取内部值
        return p.data[self_idx];
    } else if (t == 4 || t == 5) { // PressureInlet/Outlet → Dirichlet: 0.0
        return 0.0f;
    } else {
        return 0.0f;
    }
}

void main() {
    uint idx = gl_GlobalInvocationID.x;
    if (idx >= NX * NY) return;

    uint i = idx % NX;
    uint j = idx / NX;

    float p_left   = getPressure(int(i)-1, int(j), idx);
    float p_right  = getPressure(int(i)+1, int(j), idx);
    float p_bottom = getPressure(int(i), int(j)-1, idx);
    float p_top    = getPressure(int(i), int(j)+1, idx);

    u.data[idx] -= (p_right - p_left) * INV_2DX;
    v.data[idx] -= (p_top - p_bottom) * INV_2DX;
}
