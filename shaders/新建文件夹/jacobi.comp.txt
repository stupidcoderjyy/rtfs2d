#version 450

layout(local_size_x = 128, local_size_y = 1, local_size_z = 1) in;

const uint NX = 128u;
const uint NY = 128u;

// 边界偏移常量
const uint TOP_START    = 0u;
const uint BOTTOM_START = NX;
const uint LEFT_START   = 2u * NX;
const uint RIGHT_START  = 2u * NX + NY;

layout(std430, binding = 0) buffer CurrentField { float data[]; } cur;
layout(std430, binding = 1) buffer SourceField { float data[]; } src;
layout(std430, binding = 3) buffer NewField { float data[]; } newF;
layout(std430, binding = 5) buffer BCType  { int data[]; } bc_type;
layout(std430, binding = 6) buffer BCVelU  { float data[]; } bc_vel_u;   // 本求解器中未使用，仅为保持绑定对齐
layout(std430, binding = 7) buffer BCVelV  { float data[]; } bc_vel_v;   // 本求解器中未使用

layout(push_constant) uniform Params {
    float alpha;
    float beta;
} pc;

float getNeighbor(int i, int j, uint self_idx) {
    if (i >= 0 && i < int(NX) && j >= 0 && j < int(NY)) {
        uint idx = uint(i) + uint(j) * NX;
        return cur.data[idx];
    }
    // 越界，查询边界类型
    int dir = -1;
    uint local = 0;
    if (i < 0) { dir = 0; local = uint(j); }          // 左边界
    else if (i >= int(NX)) { dir = 2; local = uint(j); } // 右边界
    else if (j < 0) { dir = 3; local = uint(i); }      // 下边界
    else { dir = 1; local = uint(i); }                 // 上边界

    uint offset = (dir == 0) ? LEFT_START : (dir == 2) ? RIGHT_START : (dir == 1) ? TOP_START : BOTTOM_START;
    int t = bc_type.data[offset + local];
    if (t == 1) {               // NoSlipWall → Neumann: 镜像（取自身）
        return cur.data[self_idx];
    } else if (t == 4 || t == 5) { // PressureInlet/Outlet → Dirichlet: 0.0
        return 0.0f;
    } else {
        return cur.data[self_idx];
    }
}

void main() {
    uint idx = gl_GlobalInvocationID.x;
    if (idx >= NX * NY) return;

    uint i = idx % NX;
    uint j = idx / NX;

    float left   = getNeighbor(int(i)-1, int(j), idx);
    float right  = getNeighbor(int(i)+1, int(j), idx);
    float bottom = getNeighbor(int(i), int(j)-1, idx);
    float top    = getNeighbor(int(i), int(j)+1, idx);

    float b = src.data[idx];
    float newVal = (left + right + bottom + top + pc.alpha * b) / pc.beta;
    newF.data[idx] = newVal;
}
